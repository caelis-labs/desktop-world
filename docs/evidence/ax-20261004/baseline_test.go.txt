package acceptance_test
import (
 "context"
 "crypto/sha256"
 "encoding/hex"
 dw "github.com/caelis-labs/desktop-world"
 "github.com/caelis-labs/desktop-world/host"
 "github.com/caelis-labs/desktop-world/protocol"
 "os"
 "strings"
 "testing"
 "time"
)
func TestIssue6Baseline(t *testing.T) {
 target, err := os.ReadFile("/tmp/desktop-world-issue6-target-name"); if err != nil { t.Fatal(err) }
 ctx,cancel := context.WithTimeout(context.Background(),40*time.Second); defer cancel()
 c,err := host.Start(ctx,host.Options{Executable:os.Getenv("DW_NATIVE_HELPER_PATH")}); if err != nil { t.Fatal(err) }; defer c.Close()
 turn := "issue6-old"; if err:=c.BeginTurn(ctx,turn); err!=nil {t.Fatal(err)}
 reply,err := c.Call(ctx,turn,"inventory","observe",dw.ObserveRequest{Scope:dw.Scope{Desktop:true},Projection:dw.ProjectionSummary,Budget:dw.Budget{MaxResults:1024,MaxVisitedNodes:2048,MaxOutputBytes:1<<20,ReadDeadline:10*time.Second}}); if err!=nil || reply.Error!=nil {t.Fatalf("inventory %v %+v",err,reply.Error)}
 var inventory dw.Observation; if err:=protocol.Decode(reply.Result,&inventory); err!=nil {t.Fatal(err)}
 var window,app dw.Ref
 for _,o:=range inventory.Objects {if o.Kind==dw.KindWindow && o.Name.Value!=nil && strings.HasPrefix(*o.Name.Value,"United States - Wikipedia") {window,app=o.Ref,o.App}}
 if window=="" {t.Fatal("Wikipedia window absent")}; if err:=c.Grant(ctx,turn,app); err!=nil {t.Fatal(err)}
 name:=string(target); hash:=sha256.Sum256(target)
 req:=dw.ObserveRequest{Scope:dw.Scope{Refs:[]dw.Ref{window}},Projection:dw.ProjectionOutline,Fields:[]string{"role","name"},Match:&dw.Locator{Within:window,Role:"link",NameEquals:&name},Budget:dw.Budget{MaxDepth:30,MaxVisitedNodes:10000,MaxResults:8,MaxOutputBytes:4096,ReadDeadline:10*time.Second}}
 start:=time.Now(); reply,err=c.Call(ctx,turn,"exact","observe",req); elapsed:=time.Since(start)
 if err!=nil || reply.Error!=nil {t.Fatalf("exact %v %+v",err,reply.Error)}
 var ob dw.Observation; if err:=protocol.Decode(reply.Result,&ob); err!=nil {t.Fatal(err)}
 bytes:=len(host.Content(reply).Content[0].Text)
 t.Logf("old_exact calls=1 visited=%d hits=%d model_bytes=%d elapsed=%s complete=%t continuation=%t sources=%v target_sha256=%s",ob.Coverage.VisitedNodes,len(ob.Objects),bytes,elapsed,ob.Coverage.Complete,ob.Coverage.Continuation!="",ob.Coverage.UnavailableSources,hex.EncodeToString(hash[:]))
 if len(ob.Objects)!=0 || ob.Coverage.Complete || ob.Coverage.Continuation!="" {t.Fatalf("unexpected old behavior: %+v",ob.Coverage)}
}
